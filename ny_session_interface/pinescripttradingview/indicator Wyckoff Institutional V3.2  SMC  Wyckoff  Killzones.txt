    while array.size(_boxes) > keepN
        box.delete(array.pop(_boxes))
        array.pop(_tops)
        array.pop(_bots)

    float top = high[1]
    float bot = low[1]
    box b = box.new(bar_index - 1, top, bar_index + obExtendBars, bot, bgcolor=color.new(color.green, 84), border_color=color.new(color.green, 10))
    array.unshift(bullZones, b)
    array.unshift(bullTop, top)
    array.unshift(bullBot, bot)

createBearOB() =>
    float top = high[1]
    float bot = low[1]
    box b = box.new(bar_index - 1, top, bar_index + obExtendBars, bot, bgcolor=color.new(color.red, 84), border_color=color.new(color.red, 10))
    array.unshift(bearZones, b)
    array.unshift(bearTop, top)
    array.unshift(bearBot, bot)

touchZone(_top, _bot) =>
    not na(_top) and not na(_bot) and high >= (_bot - tol) and low <= (_top + tol)

// FVG WITH AUTO CLEAN
//====================================================
var box[] bullFvgBoxes = array.new_box()
var float[] bullFvgTop = array.new_float()
var float[] bullFvgBot = array.new_float()

var box[] bearFvgBoxes = array.new_box()
var float[] bearFvgTop = array.new_float()
var float[] bearFvgBot = array.new_float()

bullFVG = low > high[2]

    float top = low
    float bot = high[2]
    if showFVG
        box b = box.new(bar_index - 2, top, bar_index + fvgExtendBars, bot, bgcolor=color.new(color.lime, 86), border_color=color.new(color.green, 20))
        array.unshift(bullFvgBoxes, b)
        array.unshift(bullFvgTop, top)
        array.unshift(bullFvgBot, bot)

    float top = low[2]
    float bot = high
    if showFVG
        box b = box.new(bar_index - 2, top, bar_index + fvgExtendBars, bot, bgcolor=color.new(color.red, 86), border_color=color.new(color.red, 20))
        array.unshift(bearFvgBoxes, b)
        array.unshift(bearFvgTop, top)
        array.unshift(bearFvgBot, bot)

// delete mitigated FVG

    int i = array.size(bullFvgBoxes) - 1
    while i >= 0
        float t = array.get(bullFvgTop, i)
        float b = array.get(bullFvgBot, i)
        bool mitigated = low <= b
        if mitigated
            box.delete(array.get(bullFvgBoxes, i))
            array.remove(bullFvgBoxes, i)
            array.remove(bullFvgTop, i)
            array.remove(bullFvgBot, i)

    int j = array.size(bearFvgBoxes) - 1
    while j >= 0
        float t2 = array.get(bearFvgTop, j)
        float b2 = array.get(bearFvgBot, j)

        if mitigated2
            box.delete(array.get(bearFvgBoxes, j))

            array.remove(bearFvgTop, j)
            array.remove(bearFvgBot, j)

touchBearFVG = not na(lastBearFVGTop) and high >= lastBearFVGBot and low <= lastBearFVGTop

//====================================================
// HTF BIAS

f_getBias(tf) =>
    h = request.security(syminfo.tickerid, tf, ta.highest(high, 40))
    l = request.security(syminfo.tickerid, tf, ta.lowest(low, 40))
    c = request.security(syminfo.tickerid, tf, close)
    eq = (h + l) / 2.0
    c > eq ? 1 : c < eq ? -1 : 0

biasH4  = f_getBias("240")
biasH1  = f_getBias("60")
biasM15 = f_getBias("15")
biasM5  = f_getBias("5")

htfBullBias = biasH4 == 1 and biasH1 == 1
htfBearBias = biasH4 == -1 and biasH1 == -1
ltfBullTrig = biasM15 == 1 or biasM5 == 1
ltfBearTrig = biasM15 == -1 or biasM5 == -1

bearScore += inPremiumZone ? 2 : 0

bearScore += (bearOBRetest or bearBreakerRetest) ? 2 : 0

bearScore += ((not useFVGFilter) or touchBearFVG) ? 1 : 0

bearScore += ((not useRSI) or rsi <= rsiBearMax) ? 1 : 0

bearScore += (sessionOk and ltfBearTrig) ? 1 : 0

bearScore += (sweepAsiaHigh ? 1 : 0)

bearScore += ((not useDisplacementFilter) or isBearDisplacement ? 1 : 0)

    liveTP2 := tp2Buy
    liveTrail := slBuy
    liveDir := 1
    tp1Hit := false

if sellSignal

    liveTP1 := tp1Sell
    liveTP2 := tp2Sell
    liveTrail := slSell
    liveDir := -1

    tp1Hit := true

if liveDir == -1 and not tp1Hit and low <= liveTP1
    tp1Hit := true

if useTrailing and liveDir == 1
    bool trailAllowed = trailAfterTP1 ? tp1Hit : true
    if trailAllowed
        float candidate = close - atr * trailAtrMult
        liveTrail := na(liveTrail) ? candidate : math.max(liveTrail, candidate)

if useTrailing and liveDir == -1
    bool trailAllowed = trailAfterTP1 ? tp1Hit : true
    if trailAllowed
        float candidate = close + atr * trailAtrMult
        liveTrail := na(liveTrail) ? candidate : math.min(liveTrail, candidate)

if showTradeLevels and (buySignal or sellSignal or liveDir != 0)

    line.delete(tp1Line)
    line.delete(tp2Line)

        tp1Line   := line.new(bar_index, liveTP1, bar_index + 20, liveTP1, color=color.lime, width=2)

        trailLine := line.new(bar_index, liveTrail, bar_index + 20, liveTrail, color=color.yellow, width=2)
        signalLabel := label.new(bar_index, low, directionText + "\nScore: " + str.tostring(bullScore) + "/10", style=label.style_label_up, color=color.lime, textcolor=color.black)

    if liveDir == -1
        entryLine := line.new(bar_index, liveEntry, bar_index + 20, liveEntry, color=color.white, width=2)
        slLine    := line.new(bar_index, liveSL, bar_index + 20, liveSL, color=color.red, width=2)
        tp1Line   := line.new(bar_index, liveTP1, bar_index + 20, liveTP1, color=color.lime, width=2)
        tp2Line   := line.new(bar_index, liveTP2, bar_index + 20, liveTP2, color=color.green, width=2)
        trailLine := line.new(bar_index, liveTrail, bar_index + 20, liveTrail, color=color.yellow, width=2)
        signalLabel := label.new(bar_index, high, directionText + "\nScore: " + str.tostring(bearScore) + "/10", style=label.style_label_down, color=color.red, textcolor=color.white)

biasToText(v) =>

biasToBg(v) =>
    v == 1 ? color.new(color.green, 45) : v == -1 ? color.new(color.red, 45) : color.new(color.gray, 60)

f_getBias(tf) =>
    h = request.security(syminfo.tickerid, tf, ta.highest(high, 40))
    l = request.security(syminfo.tickerid, tf, ta.lowest(low, 40))
    c = request.security(syminfo.tickerid, tf, close)
    eq = (h + l) / 2.0
    c > eq ? 1 : c < eq ? -1 : 0
